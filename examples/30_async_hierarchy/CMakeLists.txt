cmake_minimum_required(VERSION 3.21)
project(NexaUIAsyncHierarchy LANGUAGES C)
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
add_library(async_hierarchy_view STATIC hierarchy_view.c)
target_link_libraries(async_hierarchy_view PUBLIC NexaUI::NexaUI)
target_include_directories(async_hierarchy_view PUBLIC .)
add_executable(async_hierarchy_core ../../tests/test_async_hierarchy.c)
target_link_libraries(async_hierarchy_core PRIVATE async_hierarchy_view)
foreach(target async_hierarchy_view async_hierarchy_core)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME async_hierarchy_core COMMAND async_hierarchy_core)
set_tests_properties(async_hierarchy_core PROPERTIES TIMEOUT 15)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Threads REQUIRED)
    add_executable(async_hierarchy_native native.c)
    target_link_libraries(async_hierarchy_native PRIVATE async_hierarchy_view Threads::Threads)
    set_target_properties(async_hierarchy_native PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(async_hierarchy_native PRIVATE -Wall -Wextra -Wpedantic)
    option(AH_BUILD_ATSPI_CLIENT "Build the independent OS libatspi test peer" OFF)
    if(AH_BUILD_ATSPI_CLIENT)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(AH_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
        add_executable(async_hierarchy_atspi atspi_client.c)
        target_link_libraries(async_hierarchy_atspi PRIVATE PkgConfig::AH_ATSPI)
        set_target_properties(async_hierarchy_atspi PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        target_compile_options(async_hierarchy_atspi PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
