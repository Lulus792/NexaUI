cmake_minimum_required(VERSION 3.24)
project(NexaUIOriginalEditorConsumer LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This independent original Editor consumer targets Linux")
endif()
find_package(NexaUI 0.1.27 CONFIG REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
option(EDITOR_BUILD_IME_MODEL "Build the original-model atomic IME adapter and its tests" OFF)
option(EDITOR_BUILD_NATIVE_IME "Connect complete native IME transactions and preedit to the original Editor" OFF)
option(EDITOR_BUILD_ACCESSIBLE_MODEL "Build original-model scalar reads/edits and atomic selection for accessibility" OFF)
option(EDITOR_BUILD_NATIVE_ACCESSIBILITY "Connect installed SDK document accessibility to the original Editor" OFF)
option(EDITOR_BUILD_NATIVE_CLIPBOARD "Connect native document clipboard requests to the original Editor window" OFF)
option(EDITOR_BUILD_NATIVE_PRESENTATION "Publish actual original Editor layout/caret geometry" OFF)
option(EDITOR_BUILD_NATIVE_ATTRIBUTES "Publish actual original Editor font/style/selection attributes" OFF)
if(EDITOR_BUILD_NATIVE_ATTRIBUTES)
    if(NOT EDITOR_BUILD_NATIVE_PRESENTATION)
        message(FATAL_ERROR "EDITOR_BUILD_NATIVE_ATTRIBUTES requires EDITOR_BUILD_NATIVE_PRESENTATION")
    endif()
    find_package(NexaUI 0.1.31 CONFIG REQUIRED)
endif()
set(EDITOR_PRESENTATION_TEST_FONT "" CACHE FILEPATH "Borrowed real font for original rendered-owner presentation tests")
set(EDITOR_PRESENTATION_LIGATURE_FONT "" CACHE FILEPATH "Borrowed proportional font with real Latin ligatures for required shaping coverage")
if(EDITOR_BUILD_NATIVE_PRESENTATION)
    if(NOT EDITOR_BUILD_NATIVE_ACCESSIBILITY)
        message(FATAL_ERROR "EDITOR_BUILD_NATIVE_PRESENTATION requires EDITOR_BUILD_NATIVE_ACCESSIBILITY")
    endif()
    # SDK0.1.30 must contain aa899fb's interior-caret correction or a later fix.
    find_package(NexaUI 0.1.30 CONFIG REQUIRED)
endif()
if(EDITOR_BUILD_NATIVE_CLIPBOARD)
    if(NOT EDITOR_BUILD_NATIVE_ACCESSIBILITY)
        message(FATAL_ERROR "EDITOR_BUILD_NATIVE_CLIPBOARD requires EDITOR_BUILD_NATIVE_ACCESSIBILITY")
    endif()
    find_package(NexaUI 0.1.29 CONFIG REQUIRED)
endif()
if(EDITOR_BUILD_NATIVE_ACCESSIBILITY)
    if(NOT EDITOR_BUILD_ACCESSIBLE_MODEL)
        message(FATAL_ERROR "EDITOR_BUILD_NATIVE_ACCESSIBILITY requires EDITOR_BUILD_ACCESSIBLE_MODEL")
    endif()
    find_package(NexaUI 0.1.28 CONFIG REQUIRED)
endif()
if(EDITOR_BUILD_NATIVE_IME AND NOT EDITOR_BUILD_IME_MODEL)
    message(FATAL_ERROR "EDITOR_BUILD_NATIVE_IME requires EDITOR_BUILD_IME_MODEL")
endif()
if(EDITOR_BUILD_ACCESSIBLE_MODEL AND NOT EDITOR_BUILD_IME_MODEL)
    message(FATAL_ERROR "EDITOR_BUILD_ACCESSIBLE_MODEL requires EDITOR_BUILD_IME_MODEL")
endif()
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable exported Aetheri snapshot; never the active checkout")
set(original "${AETHERI_SOURCE_DIR}/src")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_host.py"
    "${AETHERI_SOURCE_DIR}" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/prepare_host.py" "${CMAKE_CURRENT_SOURCE_DIR}/source-inputs.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/window_input.h" "${original}/editor/graphical_main.c")
set(model editor_original)
set(document "${original}/editor/document.c")
set(direct "${original}/editor/direct.c")
if(EDITOR_BUILD_IME_MODEL)
    set(model editor_ime_model)
    set(ime_generated "${generated}/ime-model")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_ime_model.py"
        "${AETHERI_SOURCE_DIR}" "${ime_generated}" COMMAND_ERROR_IS_FATAL ANY)
    set(document "${ime_generated}/document.c")
    set(direct "${ime_generated}/direct.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_ime_model.py")
endif()
set(host "${generated}/editor.c")
if(EDITOR_BUILD_ACCESSIBLE_MODEL)
    set(accessible_generated "${generated}/accessible-model")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_accessible_model.py"
        "${ime_generated}" "${accessible_generated}" COMMAND_ERROR_IS_FATAL ANY)
    set(document "${accessible_generated}/document.c")
    set(direct "${accessible_generated}/direct.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_accessible_model.py")
endif()
if(EDITOR_BUILD_NATIVE_IME)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_ime_host.py"
        "${host}" "${generated}/ime-host" COMMAND_ERROR_IS_FATAL ANY)
    set(host "${generated}/ime-host/editor.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_ime_host.py")
endif()
if(EDITOR_BUILD_NATIVE_ACCESSIBILITY)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_native_document_host.py"
        "${host}" "${generated}/native-document" COMMAND_ERROR_IS_FATAL ANY)
    set(host "${generated}/native-document/editor.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/prepare_native_document_host.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/native_document.h" "${CMAKE_CURRENT_SOURCE_DIR}/native_document.inc")
endif()
add_library(${model} STATIC "${original}/text/grapheme.c"
    "${document}" "${direct}"
    "${original}/editor/modal.c" "${original}/editor/file.c"
    "${original}/editor/workspace.c" "${original}/editor/view.c"
    "${original}/files/model.c" "${original}/files/launch.c" "${original}/files/create.c"
    "${original}/files/rename.c" "${original}/files/layout.c")
target_include_directories(${model} PUBLIC "${original}" PRIVATE "${original}/editor" "${CMAKE_CURRENT_SOURCE_DIR}")
set_source_files_properties("${original}/editor/file.c" "${original}/editor/workspace.c"
    "${original}/editor/view.c" PROPERTIES COMPILE_DEFINITIONS _GNU_SOURCE)
target_link_libraries(${model} PUBLIC Threads::Threads m)
add_executable(aetheri_editor_candidate "${host}")
target_include_directories(aetheri_editor_candidate PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(aetheri_editor_candidate PRIVATE ${model} NexaUI::NexaUI)
set(checked ${model} aetheri_editor_candidate)
enable_testing()
if(EDITOR_BUILD_NATIVE_ACCESSIBILITY)
    add_executable(editor_native_document_test test_native_document.c)
    target_include_directories(editor_native_document_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(editor_native_document_test PRIVATE EDITOR_NATIVE_DOCUMENT_HOST="${host}")
    target_link_libraries(editor_native_document_test PRIVATE ${model} NexaUI::NexaUI)
    target_link_options(editor_native_document_test PRIVATE -Wl,--wrap=malloc,--wrap=realloc)
    if(EDITOR_BUILD_NATIVE_CLIPBOARD)
        target_compile_definitions(aetheri_editor_candidate PRIVATE EDITOR_DOCUMENT_WITH_CLIPBOARD=1)
        target_compile_definitions(editor_native_document_test PRIVATE EDITOR_DOCUMENT_WITH_CLIPBOARD=1)
    endif()
    if(EDITOR_BUILD_NATIVE_IME)
        target_compile_definitions(aetheri_editor_candidate PRIVATE EDITOR_DOCUMENT_WITH_IME=1)
        target_compile_definitions(editor_native_document_test PRIVATE EDITOR_DOCUMENT_WITH_IME=1)
        target_link_libraries(editor_native_document_test PRIVATE editor_ime_batch)
    endif()
    if(EDITOR_BUILD_NATIVE_PRESENTATION)
        add_library(editor_presentation STATIC presentation.c)
        target_link_libraries(editor_presentation PUBLIC NexaUI::NexaUI m)
        foreach(target aetheri_editor_candidate editor_native_document_test)
            target_compile_definitions(${target} PRIVATE EDITOR_DOCUMENT_WITH_PRESENTATION=1)
            target_link_libraries(${target} PRIVATE editor_presentation)
        endforeach()
        list(APPEND checked editor_presentation)
        add_executable(editor_presentation_test test_presentation.c)
        target_link_libraries(editor_presentation_test PRIVATE editor_presentation)
        target_link_options(editor_presentation_test PRIVATE -Wl,--wrap=realloc)
        list(APPEND checked editor_presentation_test)
        if(EDITOR_PRESENTATION_TEST_FONT)
            add_test(NAME editor.presentation_real_layout COMMAND editor_presentation_test "${EDITOR_PRESENTATION_TEST_FONT}")
            add_test(NAME editor.presentation_original_owner COMMAND editor_native_document_test "${EDITOR_PRESENTATION_TEST_FONT}")
        endif()
        if(EDITOR_PRESENTATION_LIGATURE_FONT)
            add_test(NAME editor.presentation_real_ligature COMMAND editor_presentation_test "${EDITOR_PRESENTATION_LIGATURE_FONT}" --require-ligature)
            add_test(NAME editor.presentation_ligature_owner COMMAND editor_native_document_test "${EDITOR_PRESENTATION_LIGATURE_FONT}" --require-ligature)
        endif()
        if(EDITOR_BUILD_NATIVE_ATTRIBUTES)
            add_library(editor_attributes STATIC attributes.c)
            target_link_libraries(editor_attributes PUBLIC NexaUI::NexaUI m)
            foreach(target aetheri_editor_candidate editor_native_document_test)
                target_compile_definitions(${target} PRIVATE EDITOR_DOCUMENT_WITH_ATTRIBUTES=1)
                target_link_libraries(${target} PRIVATE editor_attributes)
            endforeach()
            add_executable(editor_attributes_test test_attributes.c)
            target_link_libraries(editor_attributes_test PRIVATE editor_attributes)
            add_test(NAME editor.attributes_contract COMMAND editor_attributes_test)
            if(EDITOR_PRESENTATION_TEST_FONT)
                add_test(NAME editor.attributes_real_font COMMAND editor_attributes_test "${EDITOR_PRESENTATION_TEST_FONT}")
            endif()
            list(APPEND checked editor_attributes editor_attributes_test)
        endif()
    endif()
    list(APPEND checked editor_native_document_test)
    add_test(NAME editor.native_document_original_owner COMMAND editor_native_document_test)
endif()
if(EDITOR_BUILD_ACCESSIBLE_MODEL)
    add_library(editor_accessible_model STATIC accessible_model.c)
    target_link_libraries(editor_accessible_model PUBLIC ${model})
    add_executable(editor_accessible_model_test test_accessible_model.c)
    target_link_libraries(editor_accessible_model_test PRIVATE editor_accessible_model)
    target_link_options(editor_accessible_model_test PRIVATE -Wl,--wrap=malloc)
    list(APPEND checked editor_accessible_model editor_accessible_model_test)
    add_test(NAME editor.accessible_original_model COMMAND editor_accessible_model_test)
endif()
if(EDITOR_BUILD_IME_MODEL)
    add_library(editor_ime_batch STATIC ime_batch.c)
    target_link_libraries(editor_ime_batch PUBLIC ${model} NexaUI::NexaUI)
    add_executable(editor_ime_batch_test test_ime_batch.c)
    target_link_libraries(editor_ime_batch_test PRIVATE editor_ime_batch)
    target_link_options(editor_ime_batch_test PRIVATE -Wl,--wrap=malloc)
    target_link_libraries(aetheri_editor_candidate PRIVATE editor_ime_batch)
    list(APPEND checked editor_ime_batch editor_ime_batch_test)
    add_test(NAME editor.complete_ime_batch COMMAND editor_ime_batch_test)
    add_executable(editor_ime_model_test test_ime_model.c)
    target_include_directories(editor_ime_model_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(editor_ime_model_test PRIVATE ${model})
    target_link_options(editor_ime_model_test PRIVATE -Wl,--wrap=malloc)
    list(APPEND checked editor_ime_model_test)
    add_test(NAME editor.atomic_ime_model COMMAND editor_ime_model_test)
endif()
foreach(name IN ITEMS grapheme document direct editor_workspace editor_save_work modal history search selection editor_file editor_view)
    add_executable(original_${name} "${AETHERI_SOURCE_DIR}/tests/test_${name}.c")
    target_link_libraries(original_${name} PRIVATE ${model})
    if(name STREQUAL "editor_view")
        target_sources(original_${name} PRIVATE "${original}/terminal/model.c")
    endif()
    if(name MATCHES "^(document|direct)$")
        target_link_options(original_${name} PRIVATE -Wl,--wrap=malloc,--wrap=calloc)
    elseif(name STREQUAL "editor_workspace")
        target_compile_definitions(original_${name} PRIVATE _GNU_SOURCE)
        target_link_options(original_${name} PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=fsync,--wrap=write)
    elseif(name STREQUAL "editor_save_work")
        target_compile_definitions(original_${name} PRIVATE _GNU_SOURCE)
        target_link_options(original_${name} PRIVATE -Wl,--wrap=fsync,--wrap=malloc,--wrap=calloc)
    elseif(name MATCHES "^(modal|history|search|selection)$")
        target_link_options(original_${name} PRIVATE -Wl,--wrap=malloc)
    elseif(name STREQUAL "editor_file")
        target_compile_definitions(original_${name} PRIVATE _GNU_SOURCE)
        target_link_options(original_${name} PRIVATE -Wl,--wrap=malloc,--wrap=write,--wrap=fsync,--wrap=renameat2,--wrap=fgetxattr,--wrap=close)
    endif()
    list(APPEND checked original_${name})
    add_test(NAME editor.original_${name} COMMAND original_${name})
endforeach()
set_tests_properties(editor.original_editor_save_work PROPERTIES TIMEOUT 45)
foreach(target IN LISTS checked)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endforeach()
