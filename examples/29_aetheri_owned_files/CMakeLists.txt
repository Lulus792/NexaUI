cmake_minimum_required(VERSION 3.21)
project(NexaUIOriginalFilesConsumer LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This original Aetheri Files consumer requires Linux")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable, independently selected Aetheri source snapshot")
set(NEXAUI_OWNED_FILES_TEST_FONT "" CACHE FILEPATH "Explicit test font")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-inputs.json" inputs)
string(JSON count LENGTH "${inputs}" files)
math(EXPR last "${count} - 1")
foreach(index RANGE 0 ${last})
    string(JSON path GET "${inputs}" files ${index} path)
    string(JSON expected GET "${inputs}" files ${index} sha256)
    if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${path}")
        message(FATAL_ERROR "Missing pinned original input: ${path}")
    endif()
    file(SHA256 "${AETHERI_SOURCE_DIR}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Original input differs from selected revision: ${path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AETHERI_SOURCE_DIR}/${path}")
endforeach()

# Build original application sources, not its project/package selection policy.
# This does not change Aetheri's exact SDK pin or compile its SDL view.
set(original_sources
    src/text/grapheme.c src/editor/document.c src/editor/file.c
    src/files/model.c src/files/launch.c src/files/create.c src/files/rename.c
    src/files/layout.c src/files/session.c src/files/nexaui_view.c)
list(TRANSFORM original_sources PREPEND "${AETHERI_SOURCE_DIR}/")
set_source_files_properties("${AETHERI_SOURCE_DIR}/src/editor/file.c" PROPERTIES COMPILE_DEFINITIONS _GNU_SOURCE)
add_library(original_files STATIC ${original_sources})
target_include_directories(original_files PUBLIC "${AETHERI_SOURCE_DIR}/src")
target_link_libraries(original_files PUBLIC NexaUI::NexaUI m)

add_executable(owned_files_view_test "${AETHERI_SOURCE_DIR}/tests/test_files_view.c")
target_link_libraries(owned_files_view_test PRIVATE original_files)
add_executable(owned_files_session_test "${AETHERI_SOURCE_DIR}/tests/test_files_session.c")
target_link_libraries(owned_files_session_test PRIVATE original_files)
target_link_options(owned_files_session_test PRIVATE -Wl,--wrap=mkdirat)
add_executable(owned_files_native "${AETHERI_SOURCE_DIR}/src/files/nexaui_main.c")
target_link_libraries(owned_files_native PRIVATE original_files Threads::Threads)
add_executable(owned_files_fault_native "${AETHERI_SOURCE_DIR}/src/files/nexaui_main.c" "${AETHERI_SOURCE_DIR}/tests/files_create_fault.c")
target_compile_definitions(owned_files_fault_native PRIVATE AETHERI_FAULT_ATTEMPTS)
target_link_options(owned_files_fault_native PRIVATE -Wl,--wrap=mkdirat,--wrap=renameat2)
target_link_libraries(owned_files_fault_native PRIVATE original_files Threads::Threads)
foreach(target original_files owned_files_view_test owned_files_session_test owned_files_native owned_files_fault_native)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endforeach()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
    if(ATSPI_FOUND)
        add_executable(owned_files_atspi_client "${AETHERI_SOURCE_DIR}/tests/files_view_atspi.c")
        target_link_libraries(owned_files_atspi_client PRIVATE PkgConfig::ATSPI)
        set_target_properties(owned_files_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        target_compile_options(owned_files_atspi_client PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
enable_testing()
add_test(NAME original_files.session COMMAND owned_files_session_test)
set_tests_properties(original_files.session PROPERTIES TIMEOUT 10)
if(NEXAUI_OWNED_FILES_TEST_FONT)
    if(NOT EXISTS "${NEXAUI_OWNED_FILES_TEST_FONT}")
        message(FATAL_ERROR "NEXAUI_OWNED_FILES_TEST_FONT must exist")
    endif()
    add_test(NAME original_files.view COMMAND owned_files_view_test "${NEXAUI_OWNED_FILES_TEST_FONT}" 1)
    add_test(NAME original_files.view_hidpi COMMAND owned_files_view_test "${NEXAUI_OWNED_FILES_TEST_FONT}" 2)
    set_tests_properties(original_files.view original_files.view_hidpi PROPERTIES TIMEOUT 30)
endif()
