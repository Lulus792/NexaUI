cmake_minimum_required(VERSION 3.24)
project(NexaUIAetheriSoftware LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri catalogue targets Linux")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri source snapshot")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../33_aetheri_recovery/check_source.py"
    "${AETHERI_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF)
add_subdirectory("${AETHERI_SOURCE_DIR}" aetheri EXCLUDE_FROM_ALL)
add_library(software_model STATIC software_model.c)
target_include_directories(software_model PUBLIC .)
target_link_libraries(software_model PUBLIC NexaUI::NexaUI aetheri_catalogue)
add_executable(software_model_test test_software_model.c)
target_link_libraries(software_model_test PRIVATE software_model)
target_link_options(software_model_test PRIVATE -Wl,--wrap=calloc,--wrap=ae_catalogue_get_info,--wrap=ae_catalogue_resolve_app)
add_library(software_worker STATIC software_worker.c)
target_link_libraries(software_worker PUBLIC software_model Threads::Threads)
add_executable(software_worker_test test_software_worker.c)
target_link_libraries(software_worker_test PRIVATE software_worker)
target_link_options(software_worker_test PRIVATE -Wl,--wrap=calloc,--wrap=sw_read,--wrap=ae_catalogue_open,--wrap=ae_catalogue_get_info,--wrap=ae_catalogue_close,--wrap=ae_catalogue_resolve_app,--wrap=sqlite3_prepare_v2)
foreach(target software_model software_model_test software_worker software_worker_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endforeach()
enable_testing()
add_test(NAME software_original COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_model.py"
    --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:software_model_test>" --work "${CMAKE_CURRENT_BINARY_DIR}/original-test")
set_tests_properties(software_original PROPERTIES TIMEOUT 180 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
add_test(NAME software_declared_maximum COMMAND software_model_test large)
set_tests_properties(software_declared_maximum PROPERTIES TIMEOUT 60)
add_test(NAME software_worker_original COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_worker.py"
    --source "${AETHERI_SOURCE_DIR}" --model "$<TARGET_FILE:software_model_test>" --worker "$<TARGET_FILE:software_worker_test>" --work "${CMAKE_CURRENT_BINARY_DIR}/worker-test")
set_tests_properties(software_worker_original PROPERTIES TIMEOUT 180 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
