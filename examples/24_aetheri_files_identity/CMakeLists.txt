cmake_minimum_required(VERSION 3.21)
project(NexaUIAetheriFilesIdentity LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri Files model experiment requires Linux")
endif()
find_package(NexaUI 0.1.15 CONFIG REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable Aetheri source snapshot")
foreach(path src/files/model.c src/files/model.h src/text/grapheme.c src/text/grapheme.h src/text/grapheme_data.h)
    if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${path}")
        message(FATAL_ERROR "Missing ${path} in the selected Aetheri snapshot")
    endif()
endforeach()
add_library(aetheri_files_model STATIC "${AETHERI_SOURCE_DIR}/src/files/model.c"
    "${AETHERI_SOURCE_DIR}/src/text/grapheme.c")
target_include_directories(aetheri_files_model PUBLIC "${AETHERI_SOURCE_DIR}/src")
add_executable(aetheri_files_identity_test intent.c test_identity.c)
target_link_libraries(aetheri_files_identity_test PRIVATE aetheri_files_model NexaUI::NexaUI)
add_executable(aetheri_files_scene_test scene.c intent.c test_scene.c)
target_link_libraries(aetheri_files_scene_test PRIVATE aetheri_files_model NexaUI::NexaUI)
set_target_properties(aetheri_files_model aetheri_files_identity_test aetheri_files_scene_test PROPERTIES
    C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
