cmake_minimum_required(VERSION 3.21)
project(NexaUIExternalContent LANGUAGES C)
find_package(NexaUI 0.1.33 CONFIG REQUIRED)
add_library(content_surface STATIC surface.c input.c)
target_link_libraries(content_surface PUBLIC NexaUI::NexaUI)
target_include_directories(content_surface PUBLIC .)
set_target_properties(content_surface PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
include(CTest)
option(NEXAUI_CONTENT_BUILD_PNG "Build the standalone PNG producer using libpng" OFF)
if(NEXAUI_CONTENT_BUILD_PNG)
    find_package(PNG 1.6 REQUIRED)
    add_library(content_png STATIC png_frame.c)
    target_link_libraries(content_png PUBLIC content_surface PRIVATE PNG::PNG)
    set_target_properties(content_png PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(BUILD_TESTING)
        add_executable(content_png_test test_png_frame.c)
        target_link_libraries(content_png_test PRIVATE content_png PNG::PNG)
        set_target_properties(content_png_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        add_test(NAME content_png_frame COMMAND content_png_test)
        set_tests_properties(content_png_frame PROPERTIES TIMEOUT 15)
    endif()
endif()
option(NEXAUI_CONTENT_BUILD_PDF "Build the Linux pdftoppm page producer" OFF)
if(NEXAUI_CONTENT_BUILD_PDF)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT TARGET content_png)
        message(FATAL_ERROR "The PDF producer requires Linux and NEXAUI_CONTENT_BUILD_PNG")
    endif()
    add_library(content_pdf STATIC pdf_frame.c)
    target_link_libraries(content_pdf PUBLIC content_png)
    set_target_properties(content_pdf PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(BUILD_TESTING)
        add_executable(content_pdf_test test_pdf_frame.c)
        target_link_libraries(content_pdf_test PRIVATE content_pdf PNG::PNG)
        set_target_properties(content_pdf_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        add_test(NAME content_pdf_frame COMMAND content_pdf_test)
        set_tests_properties(content_pdf_frame PROPERTIES TIMEOUT 15)
    endif()
endif()
option(NEXAUI_CONTENT_BUILD_PDF_LINKS "Build the separate Poppler-GLib PDF link reader" OFF)
if(NEXAUI_CONTENT_BUILD_PDF_LINKS)
    if(NOT TARGET content_pdf)
        message(FATAL_ERROR "The link reader requires NEXAUI_CONTENT_BUILD_PDF")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CONTENT_PDF_GIO REQUIRED IMPORTED_TARGET gio-2.0)
    find_path(CONTENT_POPPLER_INCLUDE_DIR poppler.h PATH_SUFFIXES poppler/glib REQUIRED)
    find_path(CONTENT_CAIRO_INCLUDE_DIR cairo.h PATH_SUFFIXES cairo REQUIRED)
    find_library(CONTENT_POPPLER_GLIB_LIBRARY poppler-glib REQUIRED)
    add_executable(content_pdf_link_reader pdf_link_reader.c)
    target_include_directories(content_pdf_link_reader SYSTEM PRIVATE
        ${CONTENT_POPPLER_INCLUDE_DIR} ${CONTENT_CAIRO_INCLUDE_DIR})
    target_link_libraries(content_pdf_link_reader PRIVATE
        ${CONTENT_POPPLER_GLIB_LIBRARY} PkgConfig::CONTENT_PDF_GIO m)
    set_target_properties(content_pdf_link_reader PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(content_surface_test test_surface.c)
    target_link_libraries(content_surface_test PRIVATE content_surface Threads::Threads)
    set_target_properties(content_surface_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(case flow damage malformed lifetime worker text)
        add_test(NAME content_surface_${case} COMMAND content_surface_test ${case})
        set_tests_properties(content_surface_${case} PROPERTIES TIMEOUT 15)
    endforeach()
    add_executable(content_input_test test_input.c)
    target_link_libraries(content_input_test PRIVATE content_surface)
    set_target_properties(content_input_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(case edges escape revoke identity failure)
        add_test(NAME content_input_${case} COMMAND content_input_test ${case})
        set_tests_properties(content_input_${case} PROPERTIES TIMEOUT 15)
    endforeach()
endif()

option(NEXAUI_CONTENT_BUILD_NATIVE_FIXTURE "Build the private Linux content fixture and independent peer" OFF)
if(NEXAUI_CONTENT_BUILD_NATIVE_FIXTURE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The private content fixture requires Linux Wayland/Xvfb/AT-SPI")
    endif()
    find_package(Threads REQUIRED)
    find_package(PkgConfig REQUIRED)
    find_package(X11 REQUIRED)
    pkg_check_modules(CONTENT_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
    add_executable(content_native_host native_host.c)
    target_link_libraries(content_native_host PRIVATE content_surface Threads::Threads)
    if(NEXAUI_CONTENT_BUILD_PNG)
        target_link_libraries(content_native_host PRIVATE content_png)
        target_compile_definitions(content_native_host PRIVATE CONTENT_NATIVE_PNG=1)
    endif()
    if(NEXAUI_CONTENT_BUILD_PDF)
        target_link_libraries(content_native_host PRIVATE content_pdf)
        target_compile_definitions(content_native_host PRIVATE CONTENT_NATIVE_PDF=1)
    endif()
    add_executable(content_atspi_peer atspi_peer.c)
    target_link_libraries(content_atspi_peer PRIVATE PkgConfig::CONTENT_ATSPI X11::X11)
    set_target_properties(content_native_host content_atspi_peer PROPERTIES
        C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
