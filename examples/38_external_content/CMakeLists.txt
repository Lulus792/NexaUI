cmake_minimum_required(VERSION 3.21)
project(NexaUIExternalContent LANGUAGES C)
find_package(NexaUI 0.1.33 CONFIG REQUIRED)
add_library(content_surface STATIC surface.c input.c)
target_link_libraries(content_surface PUBLIC NexaUI::NexaUI)
target_include_directories(content_surface PUBLIC .)
set_target_properties(content_surface PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(content_surface_test test_surface.c)
    target_link_libraries(content_surface_test PRIVATE content_surface Threads::Threads)
    set_target_properties(content_surface_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(case flow damage malformed lifetime worker)
        add_test(NAME content_surface_${case} COMMAND content_surface_test ${case})
        set_tests_properties(content_surface_${case} PROPERTIES TIMEOUT 15)
    endforeach()
    add_executable(content_input_test test_input.c)
    target_link_libraries(content_input_test PRIVATE content_surface)
    set_target_properties(content_input_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(case edges escape revoke identity failure)
        add_test(NAME content_input_${case} COMMAND content_input_test ${case})
        set_tests_properties(content_input_${case} PROPERTIES TIMEOUT 15)
    endforeach()
endif()

option(NEXAUI_CONTENT_BUILD_NATIVE_FIXTURE "Build the private Linux content fixture and independent peer" OFF)
if(NEXAUI_CONTENT_BUILD_NATIVE_FIXTURE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The private content fixture requires Linux Wayland/Xvfb/AT-SPI")
    endif()
    find_package(Threads REQUIRED)
    find_package(PkgConfig REQUIRED)
    find_package(X11 REQUIRED)
    pkg_check_modules(CONTENT_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
    add_executable(content_native_host native_host.c)
    target_link_libraries(content_native_host PRIVATE content_surface Threads::Threads)
    add_executable(content_atspi_peer atspi_peer.c)
    target_link_libraries(content_atspi_peer PRIVATE PkgConfig::CONTENT_ATSPI X11::X11)
    set_target_properties(content_native_host content_atspi_peer PROPERTIES
        C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
