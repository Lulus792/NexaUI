cmake_minimum_required(VERSION 3.24)
project(NexaUIOriginalFilesConsumer LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Files consumer targets Linux")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(NexaUI 0.1.23 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Exported, checked original Aetheri snapshot")
set(NEXAUI_EXPECTED_REVISION "" CACHE STRING "Explicit SDK source declaration selected for this experiment")
string(LENGTH "${NEXAUI_EXPECTED_REVISION}" sdk_revision_length)
if(NOT sdk_revision_length EQUAL 40 OR NOT NEXAUI_EXPECTED_REVISION MATCHES "^[0-9a-f]+$"
   OR NOT NexaUI_SOURCE_REVISION STREQUAL NEXAUI_EXPECTED_REVISION)
    message(FATAL_ERROR "Select the exact installed SDK source declaration with NEXAUI_EXPECTED_REVISION")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_source.py"
    "${AETHERI_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(original "${AETHERI_SOURCE_DIR}/src")
add_library(files_original_text STATIC "${original}/text/grapheme.c")
target_include_directories(files_original_text PUBLIC "${original}")
add_library(files_original_document STATIC "${original}/editor/document.c")
target_link_libraries(files_original_document PUBLIC files_original_text)
add_library(files_original_file STATIC "${original}/editor/file.c")
target_compile_definitions(files_original_file PRIVATE _GNU_SOURCE)
target_link_libraries(files_original_file PUBLIC files_original_document)
add_library(files_original_model STATIC "${original}/files/model.c" "${original}/files/launch.c"
    "${original}/files/create.c" "${original}/files/rename.c" "${original}/files/layout.c")
target_link_libraries(files_original_model PUBLIC files_original_text files_original_file m)
add_library(files_original_session STATIC "${original}/files/session.c")
target_link_libraries(files_original_session PUBLIC files_original_model)
add_library(files_original_view STATIC "${original}/files/nexaui_view.c")
target_link_libraries(files_original_view PUBLIC files_original_session NexaUI::NexaUI m)
add_executable(aetheri_files_candidate "${original}/files/nexaui_main.c")
target_link_libraries(aetheri_files_candidate PRIVATE files_original_view Threads::Threads)
set(checked_targets files_original_text files_original_document files_original_file
    files_original_model files_original_session files_original_view aetheri_files_candidate)
option(FILES_BUILD_TESTS "Build unchanged original session/view tests" ON)
option(FILES_BUILD_FAULT_HOST "Build the original test-only lost-reply host; never install it" OFF)
set(FILES_TEST_FONT "" CACHE FILEPATH "Borrowed local test font, not redistributed")
if(FILES_BUILD_TESTS)
    if(NOT EXISTS "${FILES_TEST_FONT}")
        message(FATAL_ERROR "FILES_BUILD_TESTS requires a readable FILES_TEST_FONT")
    endif()
    enable_testing()
    add_executable(files_session_original_test "${AETHERI_SOURCE_DIR}/tests/test_files_session.c")
    target_link_libraries(files_session_original_test PRIVATE files_original_session)
    target_link_options(files_session_original_test PRIVATE -Wl,--wrap=mkdirat)
    add_executable(files_view_original_test "${AETHERI_SOURCE_DIR}/tests/test_files_view.c")
    target_link_libraries(files_view_original_test PRIVATE files_original_view)
    list(APPEND checked_targets files_session_original_test files_view_original_test)
    add_test(NAME files_original_session COMMAND files_session_original_test)
    add_test(NAME files_original_view_1x COMMAND files_view_original_test "${FILES_TEST_FONT}" 1)
    add_test(NAME files_original_view_2x COMMAND files_view_original_test "${FILES_TEST_FONT}" 2)
    # Preserve the upstream view deadlines, rather than declaring slower variants accepted.
    set_tests_properties(files_original_view_1x files_original_view_2x PROPERTIES TIMEOUT 30)
endif()
if(FILES_BUILD_FAULT_HOST)
    add_executable(aetheri_files_fault_candidate "${original}/files/nexaui_main.c"
        "${AETHERI_SOURCE_DIR}/tests/files_create_fault.c")
    target_compile_definitions(aetheri_files_fault_candidate PRIVATE AETHERI_FAULT_ATTEMPTS)
    target_link_libraries(aetheri_files_fault_candidate PRIVATE files_original_view Threads::Threads)
    target_link_options(aetheri_files_fault_candidate PRIVATE -Wl,--wrap=mkdirat,--wrap=renameat2)
    list(APPEND checked_targets aetheri_files_fault_candidate)
endif()
foreach(target IN LISTS checked_targets)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endforeach()
