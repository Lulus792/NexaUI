cmake_minimum_required(VERSION 3.21)
project(NexaUIAetheriTerminalView LANGUAGES C)
find_package(NexaUI 0.1.15 CONFIG REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable Aetheri source snapshot")
if(NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.h" OR
   NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/paste.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/paste.h")
    message(FATAL_ERROR "Set AETHERI_SOURCE_DIR to an Aetheri source snapshot")
endif()
add_library(aetheri_terminal_model STATIC "${AETHERI_SOURCE_DIR}/src/terminal/model.c" "${AETHERI_SOURCE_DIR}/src/terminal/paste.c")
target_include_directories(aetheri_terminal_model PUBLIC "${AETHERI_SOURCE_DIR}/src/terminal")
add_library(aetheri_terminal_view STATIC view.c)
target_link_libraries(aetheri_terminal_view PUBLIC NexaUI::NexaUI aetheri_terminal_model)
add_executable(aetheri_terminal_view_test test_view.c)
target_link_libraries(aetheri_terminal_view_test PRIVATE aetheri_terminal_view)
add_executable(aetheri_terminal_selection_test test_selection.c)
target_link_libraries(aetheri_terminal_selection_test PRIVATE aetheri_terminal_view)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(aetheri_terminal_paste STATIC paste_flow.c paste_review.c)
    target_link_libraries(aetheri_terminal_paste PUBLIC NexaUI::NexaUI aetheri_terminal_model)
    set_target_properties(aetheri_terminal_paste PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_paste_test test_paste.c)
    target_link_libraries(aetheri_terminal_paste_test PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_paste_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_review_test test_review.c)
    target_link_libraries(aetheri_terminal_review_test PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_review_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_native_paste main_native_paste.c)
    target_link_libraries(aetheri_terminal_native_paste PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_native_paste PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    set(NEXAUI_FIXTURE_SOURCE_DIR "" CACHE PATH "Optional NexaUI checkout for controlled Wayland fixture")
    set(NEXAUI_FIXTURE_PROTOCOL_DIR "" CACHE PATH "Optional NexaUI build's generated Wayland protocols")
    if(NEXAUI_FIXTURE_SOURCE_DIR AND NEXAUI_FIXTURE_PROTOCOL_DIR)
        find_package(Threads REQUIRED)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(WAYLAND_FIXTURE REQUIRED IMPORTED_TARGET wayland-server wayland-client)
        add_executable(aetheri_terminal_paste_wayland_test test_paste_wayland.c
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}/xdg-shell-protocol.c"
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}/text-input-unstable-v3-protocol.c")
        target_include_directories(aetheri_terminal_paste_wayland_test PRIVATE
            "${NEXAUI_FIXTURE_SOURCE_DIR}/tests" "${NEXAUI_FIXTURE_SOURCE_DIR}/platform/linux"
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}")
        target_link_libraries(aetheri_terminal_paste_wayland_test PRIVATE
            aetheri_terminal_paste Threads::Threads PkgConfig::WAYLAND_FIXTURE)
        set_target_properties(aetheri_terminal_paste_wayland_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endif()
    add_executable(aetheri_terminal_ax_host main_ax.c)
    target_link_libraries(aetheri_terminal_ax_host PRIVATE aetheri_terminal_view)
    add_executable(aetheri_terminal_review_ax_host main_review_ax.c)
    target_link_libraries(aetheri_terminal_review_ax_host PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_review_ax_host PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
        if(ATSPI_FOUND)
            add_executable(aetheri_terminal_atspi_client atspi_client.c)
            target_link_libraries(aetheri_terminal_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(aetheri_terminal_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            add_executable(aetheri_terminal_review_atspi_client review_atspi_client.c)
            target_link_libraries(aetheri_terminal_review_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(aetheri_terminal_review_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        endif()
    endif()
    set_target_properties(aetheri_terminal_ax_host PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
foreach(target aetheri_terminal_model aetheri_terminal_view aetheri_terminal_view_test aetheri_terminal_selection_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endforeach()
