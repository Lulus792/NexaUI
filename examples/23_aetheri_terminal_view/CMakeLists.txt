cmake_minimum_required(VERSION 3.21)
project(NexaUIAetheriTerminalView LANGUAGES C)
find_package(NexaUI 0.1.15 CONFIG REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable Aetheri source snapshot")
if(NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.h")
    message(FATAL_ERROR "Set AETHERI_SOURCE_DIR to an Aetheri source snapshot")
endif()
add_library(aetheri_terminal_model STATIC "${AETHERI_SOURCE_DIR}/src/terminal/model.c")
target_include_directories(aetheri_terminal_model PUBLIC "${AETHERI_SOURCE_DIR}/src/terminal")
add_library(aetheri_terminal_view STATIC view.c)
target_link_libraries(aetheri_terminal_view PUBLIC NexaUI::NexaUI aetheri_terminal_model)
add_executable(aetheri_terminal_view_test test_view.c)
target_link_libraries(aetheri_terminal_view_test PRIVATE aetheri_terminal_view)
add_executable(aetheri_terminal_selection_test test_selection.c)
target_link_libraries(aetheri_terminal_selection_test PRIVATE aetheri_terminal_view)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(aetheri_terminal_ax_host main_ax.c)
    target_link_libraries(aetheri_terminal_ax_host PRIVATE aetheri_terminal_view)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
        if(ATSPI_FOUND)
            add_executable(aetheri_terminal_atspi_client atspi_client.c)
            target_link_libraries(aetheri_terminal_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(aetheri_terminal_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        endif()
    endif()
    set_target_properties(aetheri_terminal_ax_host PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
foreach(target aetheri_terminal_model aetheri_terminal_view aetheri_terminal_view_test aetheri_terminal_selection_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endforeach()
