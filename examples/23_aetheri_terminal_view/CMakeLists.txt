cmake_minimum_required(VERSION 3.21)
project(NexaUIAetheriTerminalView LANGUAGES C)
find_package(NexaUI 0.1.15 CONFIG REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable Aetheri source snapshot")
if(NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/model.h" OR
   NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/paste.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/terminal/paste.h")
    message(FATAL_ERROR "Set AETHERI_SOURCE_DIR to an Aetheri source snapshot")
endif()
add_library(aetheri_terminal_model STATIC "${AETHERI_SOURCE_DIR}/src/terminal/model.c" "${AETHERI_SOURCE_DIR}/src/terminal/paste.c")
target_include_directories(aetheri_terminal_model PUBLIC "${AETHERI_SOURCE_DIR}/src/terminal")
add_library(aetheri_terminal_view STATIC view.c)
target_link_libraries(aetheri_terminal_view PUBLIC NexaUI::NexaUI aetheri_terminal_model)
option(NEXAUI_AETHERI_IME_EXPERIMENT "Build the original terminal composition adapter experiment" OFF)
if(NEXAUI_AETHERI_IME_EXPERIMENT)
    if(NexaUI_VERSION VERSION_LESS 0.1.17)
        message(FATAL_ERROR "The terminal IME experiment requires NexaUI 0.1.17 or newer")
    endif()
    foreach(required src/terminal/composition.c src/terminal/composition.h)
        if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${required}")
            message(FATAL_ERROR "Missing ${required} in the selected Aetheri source snapshot")
        endif()
    endforeach()
    add_library(aetheri_terminal_ime STATIC ime_adapter.c "${AETHERI_SOURCE_DIR}/src/terminal/composition.c")
    target_link_libraries(aetheri_terminal_ime PUBLIC aetheri_terminal_model NexaUI::NexaUI)
    add_executable(aetheri_terminal_ime_test test_ime_adapter.c)
    target_link_libraries(aetheri_terminal_ime_test PRIVATE aetheri_terminal_ime)
    foreach(target aetheri_terminal_ime aetheri_terminal_ime_test)
        set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endforeach()
endif()
add_executable(aetheri_terminal_view_test test_view.c)
target_link_libraries(aetheri_terminal_view_test PRIVATE aetheri_terminal_view)
add_executable(aetheri_terminal_selection_test test_selection.c)
target_link_libraries(aetheri_terminal_selection_test PRIVATE aetheri_terminal_view)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    option(NEXAUI_AETHERI_PTY_EXPERIMENT "Build the external original-shell PTY experiment" OFF)
    option(NEXAUI_AETHERI_MUSL_COMPAT "Use local direct-syscall wrappers for the PTY experiment on musl" OFF)
    option(NEXAUI_AETHERI_MUSL_NORMALIZE_TERMIOS "Normalize tcgetattr output for the musl-only experiment" OFF)
    if(NEXAUI_AETHERI_MUSL_NORMALIZE_TERMIOS AND NOT NEXAUI_AETHERI_MUSL_COMPAT)
        message(FATAL_ERROR "The termios experiment requires NEXAUI_AETHERI_MUSL_COMPAT")
    endif()
    if(NEXAUI_AETHERI_PTY_EXPERIMENT)
        foreach(required src/terminal/pty.c src/terminal/pty.h src/shell/main.c src/shell/execute.c
                         src/shell/jobs.c src/shell/readline.c src/shell/editor.c src/shell/completion.c
                         src/shell/parse.c src/text/grapheme.c)
            if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${required}")
                message(FATAL_ERROR "Missing ${required} in the selected Aetheri source snapshot")
            endif()
        endforeach()
        add_library(aetheri_original_pty STATIC "${AETHERI_SOURCE_DIR}/src/terminal/pty.c")
        target_include_directories(aetheri_original_pty PUBLIC "${AETHERI_SOURCE_DIR}/src/terminal")
        add_executable(aetheri_original_shell
            "${AETHERI_SOURCE_DIR}/src/shell/main.c" "${AETHERI_SOURCE_DIR}/src/shell/execute.c"
            "${AETHERI_SOURCE_DIR}/src/shell/jobs.c" "${AETHERI_SOURCE_DIR}/src/shell/readline.c"
            "${AETHERI_SOURCE_DIR}/src/shell/editor.c" "${AETHERI_SOURCE_DIR}/src/shell/completion.c"
            "${AETHERI_SOURCE_DIR}/src/shell/parse.c" "${AETHERI_SOURCE_DIR}/src/text/grapheme.c")
        target_include_directories(aetheri_original_shell PRIVATE "${AETHERI_SOURCE_DIR}/src")
        if(NEXAUI_AETHERI_MUSL_COMPAT)
            add_library(aetheri_pty_musl_compat STATIC musl_syscalls.c)
            set_target_properties(aetheri_pty_musl_compat PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            foreach(target aetheri_original_pty aetheri_original_shell)
                target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/musl_syscalls.h")
                target_link_libraries(${target} PRIVATE aetheri_pty_musl_compat)
            endforeach()
        endif()
        add_executable(aetheri_terminal_pty_view_test test_pty_view.c)
        target_link_libraries(aetheri_terminal_pty_view_test PRIVATE aetheri_terminal_view aetheri_original_pty)
        if(NEXAUI_AETHERI_MUSL_NORMALIZE_TERMIOS)
            target_compile_definitions(aetheri_pty_musl_compat PRIVATE NEXAUI_NORMALIZE_TERMIOS=1)
            target_link_options(aetheri_original_shell PRIVATE -Wl,--wrap=tcgetattr)
            target_link_options(aetheri_terminal_pty_view_test PRIVATE -Wl,--wrap=tcgetattr)
        endif()
        foreach(target aetheri_original_pty aetheri_original_shell aetheri_terminal_pty_view_test)
            set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        endforeach()
    endif()
    add_library(aetheri_terminal_paste STATIC paste_flow.c paste_review.c)
    target_link_libraries(aetheri_terminal_paste PUBLIC NexaUI::NexaUI aetheri_terminal_model)
    set_target_properties(aetheri_terminal_paste PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_paste_test test_paste.c)
    target_link_libraries(aetheri_terminal_paste_test PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_paste_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_review_test test_review.c)
    target_link_libraries(aetheri_terminal_review_test PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_review_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_executable(aetheri_terminal_native_paste main_native_paste.c)
    target_link_libraries(aetheri_terminal_native_paste PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_native_paste PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(NEXAUI_AETHERI_PTY_EXPERIMENT)
        add_executable(aetheri_terminal_native_paste_pty main_native_paste.c)
        target_compile_definitions(aetheri_terminal_native_paste_pty PRIVATE NEXAUI_TEST_PASTE_PTY=1)
        target_link_libraries(aetheri_terminal_native_paste_pty PRIVATE aetheri_terminal_paste aetheri_original_pty)
        if(NEXAUI_AETHERI_MUSL_NORMALIZE_TERMIOS)
            target_link_options(aetheri_terminal_native_paste_pty PRIVATE -Wl,--wrap=tcgetattr)
        endif()
        set_target_properties(aetheri_terminal_native_paste_pty PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endif()
    add_executable(aetheri_terminal_native_copy main_native_copy.c)
    target_link_libraries(aetheri_terminal_native_copy PRIVATE aetheri_terminal_view)
    set_target_properties(aetheri_terminal_native_copy PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    set(NEXAUI_FIXTURE_SOURCE_DIR "" CACHE PATH "Optional NexaUI checkout for controlled Wayland fixture")
    set(NEXAUI_FIXTURE_PROTOCOL_DIR "" CACHE PATH "Optional NexaUI build's generated Wayland protocols")
    if(NEXAUI_FIXTURE_SOURCE_DIR AND NEXAUI_FIXTURE_PROTOCOL_DIR)
        find_package(Threads REQUIRED)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(WAYLAND_FIXTURE REQUIRED IMPORTED_TARGET wayland-server wayland-client)
        add_executable(aetheri_terminal_paste_wayland_test test_paste_wayland.c
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}/xdg-shell-protocol.c"
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}/text-input-unstable-v3-protocol.c")
        target_include_directories(aetheri_terminal_paste_wayland_test PRIVATE
            "${NEXAUI_FIXTURE_SOURCE_DIR}/tests" "${NEXAUI_FIXTURE_SOURCE_DIR}/platform/linux"
            "${NEXAUI_FIXTURE_PROTOCOL_DIR}")
        target_link_libraries(aetheri_terminal_paste_wayland_test PRIVATE
            aetheri_terminal_paste Threads::Threads PkgConfig::WAYLAND_FIXTURE)
        set_target_properties(aetheri_terminal_paste_wayland_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if(NEXAUI_AETHERI_IME_EXPERIMENT)
            add_executable(aetheri_terminal_ime_wayland_test test_ime_wayland.c
                "${NEXAUI_FIXTURE_PROTOCOL_DIR}/xdg-shell-protocol.c"
                "${NEXAUI_FIXTURE_PROTOCOL_DIR}/text-input-unstable-v3-protocol.c")
            target_include_directories(aetheri_terminal_ime_wayland_test PRIVATE
                "${NEXAUI_FIXTURE_SOURCE_DIR}/tests" "${NEXAUI_FIXTURE_SOURCE_DIR}/platform/linux"
                "${NEXAUI_FIXTURE_PROTOCOL_DIR}")
            target_link_libraries(aetheri_terminal_ime_wayland_test PRIVATE
                aetheri_terminal_ime Threads::Threads PkgConfig::WAYLAND_FIXTURE)
            set_target_properties(aetheri_terminal_ime_wayland_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            if(NEXAUI_AETHERI_PTY_EXPERIMENT)
                add_executable(aetheri_terminal_ime_pty_test test_ime_pty_wayland.c
                    "${NEXAUI_FIXTURE_PROTOCOL_DIR}/xdg-shell-protocol.c"
                    "${NEXAUI_FIXTURE_PROTOCOL_DIR}/text-input-unstable-v3-protocol.c")
                target_include_directories(aetheri_terminal_ime_pty_test PRIVATE
                    "${NEXAUI_FIXTURE_SOURCE_DIR}/tests" "${NEXAUI_FIXTURE_SOURCE_DIR}/platform/linux"
                    "${NEXAUI_FIXTURE_PROTOCOL_DIR}")
                target_link_libraries(aetheri_terminal_ime_pty_test PRIVATE
                    aetheri_terminal_ime aetheri_original_pty Threads::Threads PkgConfig::WAYLAND_FIXTURE)
                if(NEXAUI_AETHERI_MUSL_NORMALIZE_TERMIOS)
                    target_link_options(aetheri_terminal_ime_pty_test PRIVATE -Wl,--wrap=tcgetattr)
                endif()
                set_target_properties(aetheri_terminal_ime_pty_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            endif()
        endif()
    endif()
    add_executable(aetheri_terminal_ax_host main_ax.c)
    target_link_libraries(aetheri_terminal_ax_host PRIVATE aetheri_terminal_view)
    add_executable(aetheri_terminal_review_ax_host main_review_ax.c)
    target_link_libraries(aetheri_terminal_review_ax_host PRIVATE aetheri_terminal_paste)
    set_target_properties(aetheri_terminal_review_ax_host PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
        if(ATSPI_FOUND)
            add_executable(aetheri_terminal_atspi_client atspi_client.c)
            target_link_libraries(aetheri_terminal_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(aetheri_terminal_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            add_executable(aetheri_terminal_review_atspi_client review_atspi_client.c)
            target_link_libraries(aetheri_terminal_review_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(aetheri_terminal_review_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        endif()
    endif()
    set_target_properties(aetheri_terminal_ax_host PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
foreach(target aetheri_terminal_model aetheri_terminal_view aetheri_terminal_view_test aetheri_terminal_selection_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endforeach()
