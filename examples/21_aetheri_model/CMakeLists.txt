cmake_minimum_required(VERSION 3.21)
project(NexaUIAetheriSettingsModel LANGUAGES C)
find_package(NexaUI 0.1.5 CONFIG REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Aetheri source snapshot containing src/settings/model.c")
if(NOT EXISTS "${AETHERI_SOURCE_DIR}/src/settings/model.c" OR NOT EXISTS "${AETHERI_SOURCE_DIR}/src/settings/model.h")
    message(FATAL_ERROR "Set AETHERI_SOURCE_DIR to the desired Aetheri source snapshot")
endif()
add_library(aetheri_model STATIC "${AETHERI_SOURCE_DIR}/src/settings/model.c")
target_include_directories(aetheri_model PUBLIC
    "${AETHERI_SOURCE_DIR}/src/settings" "${AETHERI_SOURCE_DIR}/src")
# Current models validate sound observations with their own UTF-8 helper.
# Older model-only exports remain usable without a text module.
if(EXISTS "${AETHERI_SOURCE_DIR}/src/text/grapheme.c")
    target_sources(aetheri_model PRIVATE "${AETHERI_SOURCE_DIR}/src/text/grapheme.c")
endif()
target_compile_features(aetheri_model PUBLIC c_std_17)
add_library(aetheri_view STATIC view.c ../common/text_wrap.c)
target_include_directories(aetheri_view PUBLIC ../common)
target_link_libraries(aetheri_view PUBLIC NexaUI::NexaUI aetheri_model)
target_compile_features(aetheri_view PUBLIC c_std_17)
add_executable(aetheri_model_view main.c)
target_link_libraries(aetheri_model_view PRIVATE aetheri_view)
add_executable(aetheri_model_view_test test_view.c)
target_link_libraries(aetheri_model_view_test PRIVATE aetheri_view)
add_executable(aetheri_model_text_wrap_test test_text_wrap.c)
target_link_libraries(aetheri_model_text_wrap_test PRIVATE aetheri_view)
option(NEXAUI_MODEL_BUILD_SOUND_TEST "Test preservation of a current original Sound model across view replacement" OFF)
if(NEXAUI_MODEL_BUILD_SOUND_TEST)
    add_executable(aetheri_model_sound_state_test test_sound_state.c)
    target_link_libraries(aetheri_model_sound_state_test PRIVATE aetheri_view)
    set_target_properties(aetheri_model_sound_state_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
option(NEXAUI_MODEL_BUILD_ORIGINAL_VIEW "Build Aetheri's supplied original Settings/Sound view and Linux host" OFF)
if(NEXAUI_MODEL_BUILD_ORIGINAL_VIEW)
    find_package(NexaUI 0.1.31 CONFIG REQUIRED)
    foreach(source src/settings/nexaui_view.c src/settings/nexaui_view.h tests/test_settings_view.c)
        if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${source}")
            message(FATAL_ERROR "The original Settings view requires ${AETHERI_SOURCE_DIR}/${source}")
        endif()
    endforeach()
    add_library(aetheri_original_view STATIC "${AETHERI_SOURCE_DIR}/src/settings/nexaui_view.c")
    target_link_libraries(aetheri_original_view PUBLIC NexaUI::NexaUI aetheri_model)
    if(NOT WIN32)
        target_link_libraries(aetheri_original_view PUBLIC m)
    endif()
    add_executable(aetheri_original_view_test "${AETHERI_SOURCE_DIR}/tests/test_settings_view.c")
    target_link_libraries(aetheri_original_view_test PRIVATE aetheri_original_view)
    add_executable(aetheri_original_sound_test test_original_sound.c)
    target_link_libraries(aetheri_original_sound_test PRIVATE aetheri_original_view)
    set_target_properties(aetheri_original_view aetheri_original_view_test aetheri_original_sound_test PROPERTIES
        C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        foreach(source src/settings/nexaui_main.c src/settings/nexaui_main.h)
            if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${source}")
                message(FATAL_ERROR "The original Linux host requires ${AETHERI_SOURCE_DIR}/${source}")
            endif()
        endforeach()
        add_executable(aetheri_original_settings "${AETHERI_SOURCE_DIR}/src/settings/nexaui_main.c")
        target_link_libraries(aetheri_original_settings PRIVATE aetheri_original_view)
        set_target_properties(aetheri_original_settings PROPERTIES
            C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endif()
endif()
option(NEXAUI_MODEL_BUILD_BENCHMARK "Build the optional Settings layout benchmark" OFF)
if(NEXAUI_MODEL_BUILD_BENCHMARK)
    add_executable(aetheri_model_view_benchmark benchmark_view.c)
    target_link_libraries(aetheri_model_view_benchmark PRIVATE aetheri_view)
    set_target_properties(aetheri_model_view_benchmark PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
foreach(target aetheri_model aetheri_view aetheri_model_view aetheri_model_view_test aetheri_model_text_wrap_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endforeach()
