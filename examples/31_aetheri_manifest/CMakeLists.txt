cmake_minimum_required(VERSION 3.24)
project(NexaUIAetheriManifest LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri core targets Linux")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri source snapshot")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_source.py"
    "${AETHERI_SOURCE_DIR}" RESULT_VARIABLE original_status COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF)
add_subdirectory("${AETHERI_SOURCE_DIR}" aetheri EXCLUDE_FROM_ALL)
add_library(manifest_hierarchy_view STATIC ../30_async_hierarchy/hierarchy_view.c ../common/text_wrap.c)
target_include_directories(manifest_hierarchy_view PUBLIC ../30_async_hierarchy)
target_link_libraries(manifest_hierarchy_view PUBLIC NexaUI::NexaUI)
add_library(manifest_source STATIC manifest_source.c source_binding.c)
target_include_directories(manifest_source PUBLIC .)
target_link_libraries(manifest_source PUBLIC manifest_hierarchy_view aetheri_core)
add_executable(manifest_hierarchy_native ../30_async_hierarchy/native.c)
target_compile_definitions(manifest_hierarchy_native PRIVATE AH_MANIFEST)
target_link_libraries(manifest_hierarchy_native PRIVATE manifest_source Threads::Threads)
add_executable(manifest_source_test test_source.c)
target_link_libraries(manifest_source_test PRIVATE manifest_source)
add_executable(manifest_hierarchy_core ../../tests/test_async_hierarchy.c)
target_link_libraries(manifest_hierarchy_core PRIVATE manifest_hierarchy_view)
add_executable(manifest_source_binding_test test_source_binding.c)
target_link_libraries(manifest_source_binding_test PRIVATE manifest_source)
foreach(target manifest_hierarchy_view manifest_source manifest_hierarchy_native manifest_source_test manifest_hierarchy_core manifest_source_binding_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_test(NAME manifest_source_binding COMMAND manifest_source_binding_test)
set_tests_properties(manifest_source_binding PROPERTIES TIMEOUT 15)
add_test(NAME manifest_hierarchy_core COMMAND manifest_hierarchy_core)
set_tests_properties(manifest_hierarchy_core PROPERTIES TIMEOUT 15)
# The original encoder creates test data; no second wire-format implementation.
add_test(NAME manifest_source COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_source.py"
    --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:manifest_source_test>"
    --work "${CMAKE_CURRENT_BINARY_DIR}/source-test")
set_tests_properties(manifest_source PROPERTIES TIMEOUT 25)
set(AH_TEST_FONT "" CACHE FILEPATH "Optional real font for complete wrapped-row verification")
if(AH_TEST_FONT)
    add_test(NAME manifest_source_real_font COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_source.py"
        --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:manifest_source_test>"
        --work "${CMAKE_CURRENT_BINARY_DIR}/source-font-test" --font "${AH_TEST_FONT}")
    set_tests_properties(manifest_source_real_font PROPERTIES TIMEOUT 25)
endif()
option(AH_BUILD_ATSPI_CLIENT "Build the independent OS libatspi test peer" OFF)
if(AH_BUILD_ATSPI_CLIENT)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(AH_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
    add_executable(manifest_hierarchy_atspi atspi_client.c)
    target_link_libraries(manifest_hierarchy_atspi PRIVATE PkgConfig::AH_ATSPI)
    set_target_properties(manifest_hierarchy_atspi PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(manifest_hierarchy_atspi PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(manifest_source_switch_atspi source_switch_atspi.c)
    target_link_libraries(manifest_source_switch_atspi PRIVATE PkgConfig::AH_ATSPI)
    set_target_properties(manifest_source_switch_atspi PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(manifest_source_switch_atspi PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
