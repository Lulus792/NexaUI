cmake_minimum_required(VERSION 3.24)
project(NexaUIAetheriRecovery LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri catalogue targets Linux")
endif()
option(RVG_BUILD_ATSPI_CLIENT "Build the independent OS libatspi peer" OFF)
if(RVG_BUILD_ATSPI_CLIENT)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(RVG_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
    add_executable(recovery_atspi atspi_client.c)
    target_link_libraries(recovery_atspi PRIVATE PkgConfig::RVG_ATSPI)
    set_target_properties(recovery_atspi PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(recovery_atspi PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(recovery_lifecycle_atspi lifecycle_atspi.c)
    target_link_libraries(recovery_lifecycle_atspi PRIVATE PkgConfig::RVG_ATSPI)
    set_target_properties(recovery_lifecycle_atspi PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(recovery_lifecycle_atspi PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri source snapshot")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_source.py"
    "${AETHERI_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF)
add_subdirectory("${AETHERI_SOURCE_DIR}" aetheri EXCLUDE_FROM_ALL)
add_library(recovery_view_model STATIC view_model.c)
target_include_directories(recovery_view_model PUBLIC .)
target_link_libraries(recovery_view_model PUBLIC NexaUI::NexaUI aetheri_catalogue)
add_executable(recovery_view_model_test test_view_model.c)
target_link_libraries(recovery_view_model_test PRIVATE recovery_view_model)
target_link_options(recovery_view_model_test PRIVATE -Wl,--wrap=calloc,--wrap=ae_recovery_inventory_free)
add_library(recovery_view STATIC recovery_view.c ../common/text_wrap.c)
target_link_libraries(recovery_view PUBLIC recovery_view_model m)
add_executable(recovery_view_test test_recovery_view.c)
target_link_libraries(recovery_view_test PRIVATE recovery_view)
target_link_options(recovery_view_test PRIVATE -Wl,--wrap=malloc)
add_library(recovery_worker STATIC recovery_worker.c)
target_link_libraries(recovery_worker PUBLIC recovery_view_model Threads::Threads)
add_executable(recovery_worker_test test_recovery_worker.c)
target_link_libraries(recovery_worker_test PRIVATE recovery_worker)
target_link_options(recovery_worker_test PRIVATE -Wl,--wrap=ae_catalogue_open,--wrap=ae_catalogue_get_info,--wrap=ae_catalogue_close,--wrap=ae_catalogue_recovery_inventory,--wrap=sqlite3_prepare_v2,--wrap=calloc)
add_executable(recovery_native native.c)
target_link_libraries(recovery_native PRIVATE recovery_view recovery_worker)
option(RVG_BUILD_LIFECYCLE_HOST "Build test-only native held-transaction host" OFF)
if(RVG_BUILD_LIFECYCLE_HOST)
    add_executable(recovery_held_native test_native_gate.c)
    target_link_libraries(recovery_held_native PRIVATE recovery_view recovery_worker)
    set_target_properties(recovery_held_native PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(recovery_held_native PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_options(recovery_held_native PRIVATE -Wl,--wrap=ae_catalogue_open,--wrap=ae_catalogue_recovery_inventory,--wrap=ae_catalogue_close,--wrap=ae_recovery_inventory_free,--wrap=sqlite3_prepare_v2,--wrap=rvg_close,--wrap=nx_window_destroy,--wrap=nx_window_poll_with_fd)
endif()
foreach(target recovery_view_model recovery_view_model_test recovery_view recovery_view_test recovery_worker recovery_worker_test recovery_native)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endforeach()
enable_testing()
add_test(NAME recovery_view_model COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_model.py"
    --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:recovery_view_model_test>"
    --work "${CMAKE_CURRENT_BINARY_DIR}/model-test")
set_tests_properties(recovery_view_model PROPERTIES TIMEOUT 45 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
add_test(NAME recovery_worker COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_worker.py"
    --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:recovery_worker_test>" --work "${CMAKE_CURRENT_BINARY_DIR}/worker-test")
set_tests_properties(recovery_worker PROPERTIES TIMEOUT 65 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
set(RVG_TEST_FONT "" CACHE FILEPATH "Real font for recovery view interaction checks")
if(RVG_TEST_FONT)
    add_test(NAME recovery_view COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_view.py"
        --source "${AETHERI_SOURCE_DIR}" --test "$<TARGET_FILE:recovery_view_test>"
        --font "${RVG_TEST_FONT}" --work "${CMAKE_CURRENT_BINARY_DIR}/view-test")
    set_tests_properties(recovery_view PROPERTIES TIMEOUT 190 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
