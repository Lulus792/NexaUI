cmake_minimum_required(VERSION 3.21)
project(NexaUIOriginalFilePicker LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri Files model requires Linux")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri Files snapshot")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-inputs.json" inputs)
string(JSON count LENGTH "${inputs}" files)
math(EXPR last "${count} - 1")
foreach(i RANGE 0 ${last})
    string(JSON path GET "${inputs}" files ${i} path)
    string(JSON expected GET "${inputs}" files ${i} sha256)
    if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${path}")
        message(FATAL_ERROR "Missing original model input: ${path}")
    endif()
    file(SHA256 "${AETHERI_SOURCE_DIR}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Original model input changed: ${path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AETHERI_SOURCE_DIR}/${path}")
endforeach()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_executable(file_picker_worker_test test_worker.c)
target_link_libraries(file_picker_worker_test PRIVATE file_picker_worker)
set_target_properties(file_picker_worker_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(file_picker_worker_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
set(original_calls create destroy refresh enter up ancestor filter hidden select entry snapshot_entry count total depth revision query ancestor_label open_selected directory)
foreach(name IN LISTS original_calls)
    target_link_options(file_picker_worker_test PRIVATE "-Wl,--wrap=ae_files_${name}")
endforeach()
