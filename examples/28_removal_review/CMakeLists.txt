cmake_minimum_required(VERSION 3.21)
project(NexaUIRemovalReview LANGUAGES C)
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
add_library(removal_review STATIC review.c)
target_link_libraries(removal_review PUBLIC NexaUI::NexaUI)
target_include_directories(removal_review PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(removal_review_test test_review.c)
target_link_libraries(removal_review_test PRIVATE removal_review)
add_executable(removal_review_large_test test_large.c)
target_link_libraries(removal_review_large_test PRIVATE removal_review)
set(NEXAUI_REMOVAL_TEST_FONT "" CACHE FILEPATH "Explicit font for removal review regression")
enable_testing()
if(NEXAUI_REMOVAL_TEST_FONT)
    if(NOT EXISTS "${NEXAUI_REMOVAL_TEST_FONT}")
        message(FATAL_ERROR "NEXAUI_REMOVAL_TEST_FONT must name an existing font")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/captures")
    add_test(NAME removal_review COMMAND removal_review_test "${NEXAUI_REMOVAL_TEST_FONT}" "${CMAKE_CURRENT_BINARY_DIR}/captures")
    set_tests_properties(removal_review PROPERTIES TIMEOUT 30)
    add_test(NAME removal_review_large COMMAND removal_review_large_test "${NEXAUI_REMOVAL_TEST_FONT}")
    set_tests_properties(removal_review_large PROPERTIES TIMEOUT 30)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(removal_review_allocation_test test_allocation.c)
    target_link_libraries(removal_review_allocation_test PRIVATE removal_review)
    target_link_options(removal_review_allocation_test PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=free)
    set_target_properties(removal_review_allocation_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(removal_review_allocation_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME removal_review_allocation COMMAND removal_review_allocation_test)
    set_tests_properties(removal_review_allocation PROPERTIES TIMEOUT 15)
    add_executable(removal_review_native main_native.c)
    target_link_libraries(removal_review_native PRIVATE removal_review)
    set_target_properties(removal_review_native PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(removal_review_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
        if(ATSPI_FOUND)
            add_executable(removal_review_atspi_client atspi_client.c)
            target_link_libraries(removal_review_atspi_client PRIVATE PkgConfig::ATSPI)
            set_target_properties(removal_review_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            target_compile_options(removal_review_atspi_client PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endif()
foreach(target removal_review removal_review_test removal_review_large_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
