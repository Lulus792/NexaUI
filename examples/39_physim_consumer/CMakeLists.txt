cmake_minimum_required(VERSION 3.21)
project(NexaUIPhysimConsumer LANGUAGES C)
find_package(NexaUI 0.1.33 CONFIG REQUIRED)
set(PHYSIM_SOURCE_DIR "" CACHE PATH "Separate verified original Physim export from prepare.py")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This initial installed Physim consumer is checked on Linux only")
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-inputs.json" manifest)
string(JSON count LENGTH "${manifest}" inputs)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
    string(JSON path GET "${manifest}" inputs ${index} path)
    string(JSON expected GET "${manifest}" inputs ${index} sha256)
    if(NOT EXISTS "${PHYSIM_SOURCE_DIR}/${path}")
        message(FATAL_ERROR "Missing pinned Physim input: ${path}; run prepare.py first")
    endif()
    file(SHA256 "${PHYSIM_SOURCE_DIR}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned Physim input mismatch: ${path}")
    endif()
endforeach()
# Original app/core sources remain external to NexaUI. Function sections let
# this bounded model consumer omit unused Physim file/run persistence APIs.
add_library(physim_original STATIC
    "${PHYSIM_SOURCE_DIR}/src/core.c" "${PHYSIM_SOURCE_DIR}/src/numerics.c"
    "${PHYSIM_SOURCE_DIR}/src/units.c" "${PHYSIM_SOURCE_DIR}/src/scene.c"
    "${PHYSIM_SOURCE_DIR}/src/math.c" "${PHYSIM_SOURCE_DIR}/app/plot_view.c"
    "${PHYSIM_SOURCE_DIR}/examples/documentation/pendulum_main.c")
target_include_directories(physim_original PUBLIC "${PHYSIM_SOURCE_DIR}/include" "${PHYSIM_SOURCE_DIR}/app")
target_compile_options(physim_original PRIVATE -ffunction-sections -fdata-sections)
target_link_libraries(physim_original PUBLIC m)
target_link_options(physim_original INTERFACE -Wl,--gc-sections)
add_library(physim_consumer STATIC model.c view.c)
target_link_libraries(physim_consumer PUBLIC physim_original NexaUI::NexaUI)
target_include_directories(physim_consumer PUBLIC .)
target_compile_options(physim_consumer PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
set_target_properties(physim_original physim_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
include(CTest)
if(BUILD_TESTING)
    set(PHYSIM_TEST_FONT "" CACHE FILEPATH "Borrowed local font for Physim consumer tests")
    if(NOT EXISTS "${PHYSIM_TEST_FONT}")
        message(FATAL_ERROR "Set PHYSIM_TEST_FONT to an existing local TrueType font")
    endif()
    add_executable(physim_original_plot_test "${PHYSIM_SOURCE_DIR}/tests/test_plot_view.c")
    target_link_libraries(physim_original_plot_test PRIVATE physim_original)
    add_executable(physim_consumer_test test_consumer.c)
    target_link_libraries(physim_consumer_test PRIVATE physim_consumer)
    target_compile_options(physim_consumer_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    set_target_properties(physim_original_plot_test physim_consumer_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_test(NAME physim_original_plot COMMAND physim_original_plot_test)
    foreach(case model view failure export)
        add_test(NAME physim_consumer_${case} COMMAND physim_consumer_test ${case} "${PHYSIM_TEST_FONT}" "${CMAKE_CURRENT_BINARY_DIR}/physim-${case}.ppm")
        set_tests_properties(physim_consumer_${case} PROPERTIES TIMEOUT 20)
    endforeach()
    set_tests_properties(physim_original_plot PROPERTIES TIMEOUT 15)
endif()

option(NEXAUI_PHYSIM_BUILD_NATIVE_FIXTURE "Build private Wayland Physim host and independent OS peer" OFF)
if(NEXAUI_PHYSIM_BUILD_NATIVE_FIXTURE)
    find_package(PkgConfig REQUIRED)
    find_package(X11 REQUIRED)
    pkg_check_modules(PHYSIM_ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
    add_executable(physim_native_host native_host.c)
    target_link_libraries(physim_native_host PRIVATE physim_consumer)
    add_executable(physim_atspi_peer atspi_peer.c)
    target_link_libraries(physim_atspi_peer PRIVATE physim_original PkgConfig::PHYSIM_ATSPI X11::X11)
    set_target_properties(physim_native_host physim_atspi_peer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(physim_native_host PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_compile_options(physim_atspi_peer PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
