cmake_minimum_required(VERSION 3.24)
project(NexaUIAetheriOperation LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The original Aetheri catalogue requires Linux")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri source snapshot")
if(NOT EXISTS "${AETHERI_SOURCE_DIR}/src/catalogue/store.h")
    message(FATAL_ERROR "Select an immutable Aetheri catalogue snapshot")
endif()
# Build the original libraries with their reviewed dependencies and schema tool.
# No live catalogue, authority service, consumer pin or Aetheri source is changed.
set(BUILD_TESTING OFF)
add_subdirectory("${AETHERI_SOURCE_DIR}" aetheri EXCLUDE_FROM_ALL)
add_library(operation_observer STATIC observer.c)
target_include_directories(operation_observer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(operation_observer PUBLIC NexaUI::NexaUI aetheri_catalogue)
add_executable(operation_observer_test test_observer.c)
target_link_libraries(operation_observer_test PRIVATE operation_observer)
target_link_options(operation_observer_test PRIVATE -Wl,--wrap=ae_catalogue_fixture_accept,--wrap=sqlite3_exec)
add_executable(operation_observer_native main_native.c)
target_link_libraries(operation_observer_native PRIVATE operation_observer)
target_link_options(operation_observer_native PRIVATE -Wl,--wrap=ae_catalogue_fixture_accept)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(ATSPI QUIET IMPORTED_TARGET atspi-2 gobject-2.0)
    if(ATSPI_FOUND)
        add_executable(operation_observer_atspi_client atspi_client.c)
        target_link_libraries(operation_observer_atspi_client PRIVATE PkgConfig::ATSPI)
        set_target_properties(operation_observer_atspi_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        target_compile_options(operation_observer_atspi_client PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
foreach(target operation_observer operation_observer_test operation_observer_native)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
