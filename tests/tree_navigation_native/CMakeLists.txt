cmake_minimum_required(VERSION 3.21)
project(NexaUITreeNativeConsumer LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This native transport fixture requires Linux Wayland")
endif()
find_package(NexaUI 0.1.23 CONFIG REQUIRED)
add_executable(tree_native_host host.c)
target_include_directories(tree_native_host PRIVATE ..)
target_link_libraries(tree_native_host PRIVATE NexaUI::NexaUI)
add_executable(tree_navigation_core ../test_tree_navigation.c)
target_link_libraries(tree_navigation_core PRIVATE NexaUI::NexaUI)
add_executable(tree_expansion_core ../test_expansion_actions.c)
target_link_libraries(tree_expansion_core PRIVATE NexaUI::NexaUI)
foreach(target tree_native_host tree_navigation_core tree_expansion_core)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(NEXAUI_TREE_NATIVE_CLIENT "Build the independent system libatspi peer" ON)
if(NEXAUI_TREE_NATIVE_CLIENT)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(TREE_ATSPI REQUIRED IMPORTED_TARGET atspi-2)
    add_executable(tree_native_client client.c)
    target_link_libraries(tree_native_client PRIVATE PkgConfig::TREE_ATSPI)
    set_target_properties(tree_native_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(tree_native_client PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(tree_action_client action_client.c)
    target_link_libraries(tree_action_client PRIVATE PkgConfig::TREE_ATSPI)
    set_target_properties(tree_action_client PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_options(tree_action_client PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME tree_navigation_core COMMAND tree_navigation_core)
set_tests_properties(tree_navigation_core PROPERTIES TIMEOUT 15)
add_test(NAME tree_expansion_core COMMAND tree_expansion_core)
set_tests_properties(tree_expansion_core PROPERTIES TIMEOUT 15)
