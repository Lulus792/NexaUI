cmake_minimum_required(VERSION 3.21)
project(NexaUIVulkanConsumer LANGUAGES C)
find_package(NexaUI CONFIG REQUIRED COMPONENTS Vulkan)
add_executable(consumer main.c)
target_compile_features(consumer PRIVATE c_std_17)
set_target_properties(consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer PRIVATE NexaUI::Vulkan)
get_target_property(_nexaui_core_type NexaUI::NexaUI TYPE)
if(_nexaui_core_type STREQUAL "SHARED_LIBRARY")
    add_library(consumer_plugin SHARED plugin.c)
    target_link_libraries(consumer_plugin PRIVATE NexaUI::Vulkan)
    target_link_libraries(consumer PRIVATE consumer_plugin)
    target_compile_definitions(consumer PRIVATE NEXAUI_CONSUMER_PLUGIN=1)
endif()
