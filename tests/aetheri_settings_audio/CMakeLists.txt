cmake_minimum_required(VERSION 3.21)
project(NexaUIOriginalSettingsAudioFixture LANGUAGES C)
set(AETHERI_SOURCE_DIR "" CACHE PATH "Immutable original Aetheri snapshot")
set(AETHERI_SYSTEMD_INCLUDE_DIR "" CACHE PATH "Systemd header SDK root (contains systemd/sd-bus.h)")
foreach(source src/settings/audio.c src/settings/audio_host.c src/audio/session_control.h
               tests/test_settings_audio.c tests/test_settings_audio_host.c)
    if(NOT EXISTS "${AETHERI_SOURCE_DIR}/${source}")
        message(FATAL_ERROR "Missing original audio fixture input: ${source}")
    endif()
endforeach()
if(NOT EXISTS "${AETHERI_SYSTEMD_INCLUDE_DIR}/systemd/sd-bus.h")
    message(FATAL_ERROR "Select the explicit systemd header SDK for the original application header")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0 gio-unix-2.0)
pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
add_library(controlled_original_audio_host STATIC
    "${AETHERI_SOURCE_DIR}/src/settings/audio.c"
    "${AETHERI_SOURCE_DIR}/src/settings/audio_host.c")
target_include_directories(controlled_original_audio_host PUBLIC "${AETHERI_SOURCE_DIR}/src")
target_include_directories(controlled_original_audio_host SYSTEM PUBLIC "${AETHERI_SYSTEMD_INCLUDE_DIR}")
target_link_libraries(controlled_original_audio_host PUBLIC aetheri_model PkgConfig::GLIB)
# Only the session-control observations/effects are supplied by test executables.
# The original Settings pump, GLib wait adapter and native entry are unchanged.
set(NEXAUI_MODEL_BUILD_ORIGINAL_VIEW ON CACHE BOOL "Original presentation")
set(NEXAUI_MODEL_BUILD_ORIGINAL_AUDIO_ENTRY ON CACHE BOOL "Original audio entry")
set(AETHERI_SETTINGS_AUDIO_HOST_TARGET controlled_original_audio_host CACHE STRING "Explicit test host")
add_subdirectory(../../examples/21_aetheri_model example21)
add_executable(original_audio_binding_test "${AETHERI_SOURCE_DIR}/tests/test_settings_audio.c")
add_executable(original_audio_events_test "${AETHERI_SOURCE_DIR}/tests/test_settings_audio_host.c")
target_link_libraries(original_audio_binding_test PRIVATE controlled_original_audio_host)
target_link_libraries(original_audio_events_test PRIVATE controlled_original_audio_host)
add_executable(controlled_audio_settings controlled_owner.c)
target_link_libraries(controlled_audio_settings PRIVATE aetheri_original_audio_entry)
add_executable(controlled_audio_atspi atspi_client.c)
target_link_libraries(controlled_audio_atspi PRIVATE PkgConfig::ATSPI)
foreach(target controlled_original_audio_host original_audio_binding_test original_audio_events_test
               controlled_audio_settings controlled_audio_atspi)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endforeach()
enable_testing()
add_test(NAME settings.audio_binding COMMAND original_audio_binding_test)
add_test(NAME settings.audio_events COMMAND original_audio_events_test)
set_tests_properties(settings.audio_binding PROPERTIES TIMEOUT 10)
set_tests_properties(settings.audio_events PROPERTIES TIMEOUT 15)
