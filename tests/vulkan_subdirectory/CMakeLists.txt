cmake_minimum_required(VERSION 3.21)
project(NexaUIVulkanSubdirectoryConsumer LANGUAGES C)
if(NOT NEXAUI_SOURCE)
    message(FATAL_ERROR "Set NEXAUI_SOURCE to the NexaUI source directory")
endif()
add_subdirectory("${NEXAUI_SOURCE}" nexaui EXCLUDE_FROM_ALL)
add_executable(consumer ../vulkan_consumer/main.c)
target_compile_features(consumer PRIVATE c_std_17)
set_target_properties(consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer PRIVATE NexaUI::Vulkan)
