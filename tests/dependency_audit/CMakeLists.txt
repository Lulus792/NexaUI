cmake_minimum_required(VERSION 3.21)
project(NexaUIDependencyAuditFixture LANGUAGES C)

if(AUDIT_CASE STREQUAL "header")
    add_library(probe STATIC forbidden.c)
elseif(AUDIT_CASE STREQUAL "header_import")
    add_library(probe STATIC forbidden_import.c)
else()
    add_library(probe STATIC clean.c)
endif()
if(AUDIT_CASE STREQUAL "link")
    target_link_libraries(probe PRIVATE SDL3)
elseif(AUDIT_CASE STREQUAL "link_option")
    target_link_options(probe PRIVATE -lSDL3)
elseif(AUDIT_CASE STREQUAL "appkit")
    target_link_libraries(probe PRIVATE "-framework AppKit" objc)
elseif(WIN32)
    target_link_libraries(probe PRIVATE user32)
else()
    target_link_libraries(probe PRIVATE m)
endif()
include(${NEXAUI_SOURCE_DIR}/cmake/NexaUIDependencyAudit.cmake)
nexaui_audit_core_target(probe)
